Praktik 1 — Knowledge Base & Forward Chaining

KB Forward Chain
praktik1_kb.py
class KnowledgeBase:
    """KB sederhana: fakta (set) + aturan (list)"""

    def __init__(self):
        self.fakta  = set()    # {(s, rel, o)}
        self.aturan = []       # [(kondisi, hasil)]

    def tambah_fakta(self, s, r, o):
        self.fakta.add((s, r, o))

    def tambah_aturan(self, cond, result):
        self.aturan.append((cond, result))

    def tanya(self, s, r, o=None):
        return [f for f in self.fakta
                if f[0]==s and f[1]==r
                and (o is None or f[2]==o)]

    def forward_chain(self):
        """Terapkan aturan sampai tidak ada fakta baru."""
        baru, i = True, 0
        while baru:
            baru = False; i += 1
            for cond, hasil in self.aturan:
                if all(c in self.fakta for c in cond):
                    if hasil not in self.fakta:
                        self.fakta.add(hasil)
                        print(f"  [iter{i}] ✅ {hasil}")
                        baru = True


# ── Build KB ──
kb = KnowledgeBase()
for f in [("Tweety","adalah","Burung"),
          ("Burung","adalah","Hewan"),
          ("Tweety","memiliki","Sayap"),
          ("Rex","adalah","Anjing"),
          ("Anjing","adalah","Mamalia"),
          ("Mamalia","adalah","Hewan")]:
    kb.tambah_fakta(*f)

kb.tambah_aturan(
    [("Tweety","adalah","Burung"),
     ("Burung","adalah","Hewan")],
    ("Tweety","adalah","Hewan"))
kb.tambah_aturan(
    [("Rex","adalah","Anjing"),
     ("Anjing","adalah","Mamalia")],
    ("Rex","adalah","Mamalia"))

print("=== Forward Chaining ===")
kb.forward_chain()
print(f"\nQuery 'Tweety=Hewan'  : {bool(kb.tanya('Tweety','adalah','Hewan'))}")
print(f"Query 'Rex=Mamalia'   : {bool(kb.tanya('Rex','adalah','Mamalia'))}")

Logika Proposisional — Operator & Tabel Kebenaran

OperatorSimbolPythonT,TT,FF,TF,F
Negasi¬Pnot PFFTT
KonjungsiP ∧ QP and QTFFF
DisjungsiP ∨ QP or QTTTF
ImplikasiP → Qnot P or QTFTT
BiimplikasiP ↔ QP == QTFFT

Modus Ponens

Jika P benar DAN P→Q benar, maka Q benar.

P = Hujan
P→Q = Hujan→Basah
∴ Q = Basah ✅

Modus Tollens

Jika ¬Q benar DAN P→Q benar, maka ¬P benar.

¬Q = Tidak basah
P→Q = Hujan→Basah
∴ ¬P = Tidak hujan ✅

De Morgan

¬(P∧Q) ≡ (¬P∨¬Q)
¬(P∨Q) ≡ (¬P∧¬Q)

Berguna menyederhanakan ekspresi logika kompleks.

Logika Rule-Based
praktik2_logika_pakar.py
import itertools

def tabel_kebenaran(vars, fn, nama="Ekspresi"):
    hdr = " | ".join(f"{v:^4}" for v in vars)
    hdr += f" ‖ {nama:^12}"
    print(hdr); print("─"*len(hdr))
    for vals in itertools.product([False,True],
                                   repeat=len(vars)):
        m = dict(zip(vars, vals))
        h = fn(m)
        row = " | ".join(f"{'T':^4}" if v else f"{'F':^4}"
                         for v in vals)
        row += f" ‖ {'T':^12}" if h else f" ‖ {'F':^12}"
        print(row)

print("── P → Q (Implikasi) ──")
tabel_kebenaran(['P','Q'],
    lambda m: not m['P'] or m['Q'], "P→Q")

print("\n── De Morgan: ¬(P∧Q) ↔ (¬P∨¬Q) ──")
tabel_kebenaran(['P','Q'],
    lambda m: (not(m['P'] and m['Q'])) == \
              (not m['P'] or not m['Q']),
    "De Morgan")

# ── Sistem Pakar Mini ──
class SistemPakarHewan:
    def __init__(self):
        self.aturan = [
            ({'berbulu','menyusui'},           'Mamalia'),
            ({'bersayap','bertelur','terbang'},'Burung'),
            ({'bersayap','bertelur','berenang'},'Penguin'),
            ({'berbulu','menyusui','terbang'}, 'Kelelawar'),
            ({'bersisik','berdarah_dingin'},   'Reptil'),
        ]
    def diagnosa(self, ciri):
        c = {x.lower() for x in ciri}
        return [k for cond,k in self.aturan
                if cond.issubset(c)] or ["?"]

sp = SistemPakarHewan()
print("\n── Sistem Pakar Hewan ──")
tests = [
    (['bersayap','bertelur','berenang'], "Penguin"),
    (['berbulu','menyusui','terbang'],   "Kelelawar"),
    (['bersisik','berdarah_dingin'],     "Reptil"),
]
for ciri, label in tests:
    hasil = sp.diagnosa(ciri)
    ok = "✅" if label in hasil else "❌"
    print(f"  {ok} {label:<12}: {hasil}")

Praktik 3 — Knowledge Graph dengan NetworkX

NetworkX Matplotlib
praktik3_knowledge_graph.py
import networkx as nx
import matplotlib.pyplot as plt

G = nx.DiGraph()

# Entitas & warna
entitas = {
    'AI':'#FFD93D','Machine Learning':'#FFD93D',
    'Deep Learning':'#FFD93D','NLP':'#5B9DFF',
    'Computer Vision':'#5B9DFF','Python':'#5BD68A',
    'NumPy':'#5BD68A','scikit-learn':'#5BD68A',
    'PyTorch':'#FF6FA5','Transformer':'#FF6FA5',
    'ChatGPT':'#FF6FA5',
}
for e in entitas: G.add_node(e)

relasi = [
    ('AI','Machine Learning','mencakup'),
    ('AI','NLP','mencakup'),
    ('AI','Computer Vision','mencakup'),
    ('Machine Learning','Deep Learning','mencakup'),
    ('Deep Learning','Transformer','menggunakan'),
    ('Transformer','ChatGPT','dasar_dari'),
    ('Python','NumPy','ekosistem'),
    ('Python','scikit-learn','ekosistem'),
    ('Python','PyTorch','ekosistem'),
    ('scikit-learn','Machine Learning','impl.'),
    ('PyTorch','Deep Learning','impl.'),
]
for s,d,r in relasi: G.add_edge(s,d,label=r)

# Gambar
fig,ax = plt.subplots(figsize=(12,7))
fig.patch.set_facecolor('#141414')
ax.set_facecolor('#141414')
pos = nx.spring_layout(G, seed=42, k=1.8)
nx.draw_networkx_nodes(G,pos,
    node_color=[entitas[n] for n in G.nodes()],
    node_size=1600,edgecolors='#141414',linewidths=2,ax=ax)
nx.draw_networkx_labels(G,pos,font_size=7,
    font_weight='bold',ax=ax)
nx.draw_networkx_edges(G,pos,edge_color='white',
    arrows=True,arrowsize=15,ax=ax,
    connectionstyle='arc3,rad=0.1')
ax.set_title("Knowledge Graph: Ekosistem AI",
    color='white',fontsize=13,fontweight='bold')
ax.axis('off')
plt.tight_layout()
plt.savefig('knowledge_graph.png',dpi=150,
    bbox_inches='tight',facecolor='#141414')
plt.show()
