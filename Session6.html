Logika Predikat — Lebih Ekspresif

Logika Proposisional (Sesi 5)

Fakta = pernyataan Benar/Salah tunggal.

"Tweety adalah burung" → P = True

❌ Tidak bisa menyatakan "semua burung bisa terbang" secara umum.

Logika Predikat (Sesi 6)

Menggunakan variabel, fungsi, dan kuantor.

∀x Burung(x) → BisaTerbang(x)
∃x Burung(x) ∧ TidakBisaTerbang(x)

✅ Bisa menyatakan aturan umum dan pengecualian.

Komponen Logika Predikat

KonstantaTweety, Rex, Jakarta
Variabelx, y, z (dapat diikat)
PredikatBurung(x), SukaMakan(x,y)
∀ (Universal)"untuk semua x berlaku..."
∃ (Eksistensial)"ada setidaknya satu x..."

Unifikasi (Unification)

Proses mencocokkan ekspresi logika dengan mengganti variabel — dasar dari inferensi predikat.

Burung(Tweety) unify Burung(x) → {x = Tweety} ✅ BisaTerbang(x) dengan x=Tweety → BisaTerbang(Tweety) ✅

Praktik 1 — Forward Chaining Lengkap

Data → Kesimpulan Inference Engine
praktik1_forward_chain.py
class InferenceEngine:
    """Mesin inferensi dengan forward & backward chaining."""

    def __init__(self):
        self.fakta   = set()
        self.aturan  = []   # [(kondisi_list, hasil, nama)]
        self.log     = []

    def assert_fact(self, *fakta):
        for f in fakta:
            self.fakta.add(f)

    def add_rule(self, kondisi, hasil, nama=""):
        self.aturan.append((kondisi, hasil, nama))

    # ── Forward Chaining ──
    def forward(self):
        """Terus terapkan aturan hingga tidak ada fakta baru."""
        self.log.clear()
        changed = True
        while changed:
            changed = False
            for cond, hasil, nama in self.aturan:
                if all(k in self.fakta for k in cond):
                    if hasil not in self.fakta:
                        self.fakta.add(hasil)
                        self.log.append(
                            f"FIRE '{nama}': {cond} → {hasil}")
                        changed = True
        return self.log

    # ── Backward Chaining ──
    def backward(self, goal, depth=0, visited=None):
        """Buktikan goal dari fakta yang ada."""
        if visited is None: visited = set()
        indent = "  " * depth
        if goal in self.fakta:
            self.log.append(f"{indent}✅ FACT: {goal}")
            return True
        if goal in visited:
            return False
        visited.add(goal)
        for cond, hasil, nama in self.aturan:
            if hasil == goal:
                self.log.append(
                    f"{indent}→ TRY rule '{nama}'")
                if all(self.backward(
                        k, depth+1, visited) for k in cond):
                    self.fakta.add(goal)
                    self.log.append(
                        f"{indent}✅ PROVED: {goal}")
                    return True
        self.log.append(f"{indent}❌ FAIL: {goal}")
        return False


# ── Demo Diagnosis Medis ──
kb = InferenceEngine()
kb.assert_fact("demam","batuk","pilek","badan_pegal")

kb.add_rule(["demam","batuk"],       "infeksi",  "R1")
kb.add_rule(["infeksi","pilek"],     "flu",      "R2")
kb.add_rule(["flu","badan_pegal"],   "flu_berat","R3")
kb.add_rule(["demam","nyeri_dada"],  "pneumonia","R4")

print("=== FORWARD CHAINING ===")
log = kb.forward()
for l in log: print(f"  {l}")
print(f"\nFakta akhir: {kb.fakta}")

Praktik 2 — Backward Chaining & Certainty Factor

Goal → Bukti CF 0–1
praktik2_backward_cf.py
class SistemPakarCF:
    """Sistem pakar dengan Certainty Factor (CF).
    CF = tingkat kepercayaan aturan (0.0–1.0).
    CF_gabungan = CF_rule × CF_fakta_terkecil
    """

    def __init__(self):
        # fakta: {nama: cf_value}
        self.fakta  = {}
        # aturan: [(cond_list, hasil, cf_rule, nama)]
        self.aturan = []

    def assert_fact(self, nama, cf=1.0):
        self.fakta[nama] = max(self.fakta.get(nama,0), cf)

    def add_rule(self, cond, hasil, cf, nama=""):
        self.aturan.append((cond, hasil, cf, nama))

    def get_cf(self, fakta_nama):
        return self.fakta.get(fakta_nama, 0.0)

    def forward_cf(self):
        changed = True
        log = []
        while changed:
            changed = False
            for cond, hasil, cf_rule, nama in self.aturan:
                cf_cond = min(self.get_cf(c) for c in cond)
                if cf_cond > 0:
                    cf_new = round(cf_rule * cf_cond, 3)
                    if cf_new > self.fakta.get(hasil, 0):
                        self.fakta[hasil] = cf_new
                        log.append(
                            f"  FIRE '{nama}': "
                            f"CF={cf_rule}×{cf_cond:.2f}"
                            f" = {cf_new}")
                        changed = True
        return log


# ── Demo Sistem Pakar Diagnosis ──
sp = SistemPakarCF()
sp.assert_fact("demam",      cf=0.9)
sp.assert_fact("batuk",      cf=0.8)
sp.assert_fact("pilek",      cf=0.7)
sp.assert_fact("badan_pegal",cf=0.6)

sp.add_rule(["demam","batuk"],     "infeksi",   0.8, "R1")
sp.add_rule(["infeksi","pilek"],   "flu",        0.9, "R2")
sp.add_rule(["flu","badan_pegal"], "flu_berat",  0.7, "R3")
sp.add_rule(["demam","batuk",
             "badan_pegal"],       "kelelahan",  0.6, "R4")

print("=== CF FORWARD CHAINING ===")
log = sp.forward_cf()
for l in log: print(l)
print("\n── Diagnosis dengan Keyakinan ──")
diagnosa = {k:v for k,v in sp.fakta.items()
            if v < 1.0 and k not in
            ["demam","batuk","pilek","badan_pegal"]}
for d,cf in sorted(diagnosa.items(),
                   key=lambda x:-x[1]):
    bar = "█"*int(cf*15)
    lvl = "TINGGI" if cf>.7 else "SEDANG" if cf>.4 else "RENDAH"
    print(f"  {d:<12} CF={cf:.3f} {bar} [{lvl}]")

Rule-Based System vs Machine Learning

Aspek🔧 Rule-Based (Sesi 5-6)🤖 Machine Learning (Sesi 2)🏆 Pilih Kapan
PengetahuanManual dari pakarOtomatis dari dataRB: pengetahuan ada, data sedikit
Explainability✅ Sangat jelas (IF-THEN)❌ Kotak hitam (NN)RB: perlu audit/regulasi
AkurasiTerbatas oleh aturan✅ Bisa sangat tinggiML: data berlabel banyak
UpdateEdit aturan manual✅ Retrain otomatisML: domain berubah cepat
Biaya awal✅ Murah (tidak butuh GPU)Mahal (data + komputasi)RB: startup/POC cepat

Kapan Rule-Based?

  • Hukum/regulasi (pajak, kredit)
  • Keamanan kritis (medis, aerospace)
  • Domain sempit, aturan stabil

Kapan ML?

  • Data berlimpah, pola kompleks
  • Gambar, suara, teks bebas
  • Akurasi melebihi pakar manusia

Tren: Neuro-Symbolic

  • LLM + Knowledge Graph = RAG
  • ML untuk persepsi → RB untuk keputusan
  • Terbaik dari keduanya

Praktik 3 — Bandingkan Rule-Based vs ML

Perbandingan Nyata
praktik3_rb_vs_ml.py
from sklearn.tree import DecisionTreeClassifier
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
import numpy as np

# ── PENDEKATAN 1: RULE-BASED ──
class RuleBasedIris:
    """Klasifikasi Iris dengan aturan IF-THEN manual."""

    def predict(self, petal_len, petal_wid):
        if petal_len <= 2.5:
            return "setosa"
        elif petal_len <= 5.0 and petal_wid <= 1.8:
            return "versicolor"
        else:
            return "virginica"

    def evaluate(self, X, y_true, names):
        preds = [self.predict(x[2], x[3]) for x in X]
        benar = sum(p==names[t] for p,t in
                    zip(preds, y_true))
        return benar / len(y_true), preds

# ── PENDEKATAN 2: ML (Decision Tree) ──
iris = load_iris()
X, y = iris.data, iris.target
X_tr,X_te,y_tr,y_te = train_test_split(
    X,y,test_size=0.3,random_state=42,stratify=y)

dt = DecisionTreeClassifier(max_depth=3,
                              random_state=42)
dt.fit(X_tr, y_tr)
acc_ml = dt.score(X_te, y_te)

# ── EVALUASI ──
rb = RuleBasedIris()
acc_rb, preds_rb = rb.evaluate(
    X_te, y_te, iris.target_names)

print("═"*45)
print(f"{'Metode':<20} {'Akurasi':<10} {'Keterangan'}")
print("─"*45)
print(f"{'Rule-Based':<20} {acc_rb:.2%}     Manual IF-THEN")
print(f"{'Decision Tree ML':<20} {acc_ml:.2%}     Belajar dari data")
print("═"*45)

# Detail 5 prediksi pertama
print("\n── 5 Prediksi Pertama ──")
y_ml = dt.predict(X_te[:5])
for i,(rb_p,ml_p,true) in enumerate(zip(
        preds_rb[:5],y_ml,y_te[:5])):
    nama_true = iris.target_names[true]
    ok_rb = "✅" if rb_p==nama_true else "❌"
    ok_ml = "✅" if iris.target_names[ml_p]==nama_true else "❌"
    print(f"  [{i}] Benar:{nama_true:<12}"
          f"RB:{ok_rb}{rb_p:<12}ML:{ok_ml}{iris.target_names[ml_p]}")
