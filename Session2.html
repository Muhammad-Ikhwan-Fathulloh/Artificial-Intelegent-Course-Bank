Komponen Agen (OOP)

  • __init__ — state internal (nama, riwayat)
  • persepsi() — membaca lingkungan
  • keputusan() — aturan if-then / model ML
  • aksi() — output ke lingkungan
  • jalankan() — siklus persepsi → aksi

Prinsip OOP yang Dipakai

  • Encapsulation — data + fungsi dalam class
  • Inheritance — agen khusus extends agen dasar
  • Polymorphism — setiap agen punya aksi berbeda
  • Abstraction — sembunyikan detail implementasi
💡 Model ML nanti akan menjadi "otak" di dalam method keputusan()

Praktik 1 — Class Agen Dasar

OOP Simple Reflex
praktik1_agen_dasar.py 
class AgenDasar:
    """Agen sederhana: persepsi → keputusan → aksi"""

    def __init__(self, nama):
        self.nama = nama
        self.riwayat = []        # memori agen

    def persepsi(self, lingkungan):
        self.riwayat.append(lingkungan)
        return lingkungan

    def keputusan(self, kondisi):
        """Simple Reflex: aturan if-then"""
        if kondisi == "kotor":
            return "bersihkan"
        elif kondisi == "bersih":
            return "diam"
        else:
            return "eksplorasi"

    def aksi(self, keputusan):
        print(f"[{self.nama}] Aksi: {keputusan.upper()}")

    def jalankan(self, kondisi_list):
        print(f"\n{'='*40}")
        print(f"  Agen '{self.nama}' mulai beroperasi")
        print(f"{'='*40}")
        for kondisi in kondisi_list:
            p = self.persepsi(kondisi)
            k = self.keputusan(p)
            self.aksi(k)
        print(f"\nRiwayat: {self.riwayat}")


# ── JALANKAN ──
ruangan = ["kotor","bersih","kotor","tidak dikenal"]
agen_vakum = AgenDasar("VacuumBot-01")
agen_vakum.jalankan(ruangan)

Praktik 2 — Inheritance & Polymorphism

Inheritance Polymorphism
praktik2_inheritance.py 
class AgenVakum(AgenDasar):
    def aksi(self, kondisi):
        return "Membersihkan 🧹" if kondisi=="kotor" \
               else "Standby ⏸"

class AgenKeamanan(AgenDasar):
    def __init__(self, nama, zona_aman):
        super().__init__(nama)
        self.zona_aman = zona_aman

    def aksi(self, kondisi):
        if kondisi not in self.zona_aman:
            return f"🚨 ALERT! '{kondisi}' tidak aman!"
        return f"✅ '{kondisi}' aman, patroli lanjut."

class AgenNavigasi(AgenDasar):
    def __init__(self, nama):
        super().__init__(nama)
        self.posisi = (0, 0)

    def aksi(self, kondisi):
        arah = {"utara":(0,1),"selatan":(0,-1),
                "timur":(1,0),"barat":(-1,0)}
        if kondisi in arah:
            dx,dy = arah[kondisi]
            self.posisi = (self.posisi[0]+dx,
                           self.posisi[1]+dy)
            return f"📍 Pindah ke {self.posisi}"
        return "🛑 Diam di tempat"

# ── POLIMORFISME ──
agen_list = [
    AgenVakum("VacBot"),
    AgenKeamanan("GuardBot", ["lobby","koridor","A"]),
    AgenNavigasi("NavBot")
]
inputs = ["kotor", "gudang", "utara"]

print("=== DEMO POLIMORFISME ===")
for agen, inp in zip(agen_list, inputs):
    hasil = agen.aksi(inp)
    print(f"[{agen.nama}] '{inp}' → {hasil}")

Praktik 3 — Pandas Dasar

DataFrame Filter GroupBy
praktik3_pandas.py 
import pandas as pd

# 1. Buat DataFrame
data = {
  'nama' : ['Andi','Budi','Citra','Deni','Eni'],
  'usia' : [21, 22, 20, 23, 21],
  'kota' : ['Jakarta','Bandung','Surabaya',
            'Medan','Jakarta'],
  'nilai': [85, 90, 78, 88, 92]
}
df = pd.DataFrame(data)
print("── DataFrame Awal ──")
print(df.to_string(index=False))

# 2. Statistik
print("\n── Statistik nilai ──")
print(df['nilai'].describe().round(2))

# 3. Filter: nilai >= 85
print("\n── Nilai >= 85 ──")
top = df[df['nilai'] >= 85][['nama','nilai']]
print(top.to_string(index=False))

# 4. Feature engineering
df['lulus'] = df['nilai'].apply(
    lambda x: '✅ Lulus' if x >= 80 else '❌ Tidak')
df['norm'] = ((df['nilai'] - df['nilai'].min()) /
              (df['nilai'].max() - df['nilai'].min())).round(2)
print("\n── Dengan Kolom Baru ──")
print(df[['nama','nilai','lulus','norm']].to_string(index=False))

# 5. GroupBy
print("\n── Rata-rata per kota ──")
grp = df.groupby('kota')['nilai'].mean().sort_values(ascending=False)
for k, v in grp.items():
    print(f"  {k:<12} {v:.1f}")

random_state=42 → hasil selalu sama setiap run (reproducible).

Praktik 4 — Model ML Pertama (KNN)

scikit-learn KNN
praktik4_knn.py 
import numpy as np
import pandas as pd
from sklearn.datasets import load_iris
from sklearn.model_selection import train_test_split
from sklearn.neighbors import KNeighborsClassifier
from sklearn.metrics import classification_report

# 1. MUAT DATA
iris = load_iris()
X, y = iris.data, iris.target
df = pd.DataFrame(X, columns=iris.feature_names)
df['spesies'] = [iris.target_names[i] for i in y]
print("── 5 Data Pertama ──")
print(df.head().to_string(index=False))
print(f"\nUkuran : {X.shape}")
print(f"Kelas  : {dict(df['spesies'].value_counts())}")

# 2. SPLIT DATA
X_train,X_test,y_train,y_test = train_test_split(
    X, y, test_size=0.3, random_state=42, stratify=y)
print(f"\nLatih: {len(X_train)} | Uji: {len(X_test)}")

# 3. LATIH MODEL
model = KNeighborsClassifier(n_neighbors=3)
model.fit(X_train, y_train)
print("✅ Model berhasil dilatih!")

# 4. EVALUASI
y_pred  = model.predict(X_test)
akurasi = model.score(X_test, y_test)
print(f"\n── Akurasi: {akurasi:.2%} ──")
print(classification_report(
    y_test, y_pred,
    target_names=iris.target_names))

# 5. PREDIKSI DATA BARU
baru = np.array([[5.1, 3.5, 1.4, 0.2]])
pred = model.predict(baru)
print(f"Prediksi bunga baru: {iris.target_names[pred[0]]}")

Praktik 5 — Agen + Model ML

Agent + ML Gabungan
praktik5_agen_ml.py 
from sklearn.datasets import load_iris
from sklearn.neighbors import KNeighborsClassifier
from sklearn.model_selection import train_test_split

class AgenKlasifikasiBunga(AgenDasar):
    """Agen dengan model ML sebagai otak keputusan"""

    def __init__(self, nama):
        super().__init__(nama)
        self.model     = None
        self.label_map = None

    def latih(self, X_train, y_train, label_map):
        """Fase pembelajaran agen"""
        self.model = KNeighborsClassifier(n_neighbors=3)
        self.model.fit(X_train, y_train)
        self.label_map = label_map
        print(f"[{self.nama}] ✅ Model dilatih!")

    def keputusan(self, fitur):
        """ML menggantikan aturan if-then"""
        if self.model is None:
            return "ERROR: belum dilatih!"
        pred = self.model.predict([fitur])[0]
        return self.label_map[pred]

    def aksi(self, hasil):
        emoji = {"setosa":"🌸","versicolor":"🌼","virginica":"🌺"}
        e = emoji.get(hasil, "🌿")
        print(f"[{self.nama}] {e} Identifikasi: {hasil.upper()}")


# ── JALANKAN ──
iris = load_iris()
X_tr,X_te,y_tr,y_te = train_test_split(
    iris.data, iris.target, test_size=0.3, random_state=42)

bot = AgenKlasifikasiBunga("FloraBot-v2")
bot.latih(X_tr, y_tr, dict(enumerate(iris.target_names)))

print("\n── Agen Mengklasifikasi 5 Sampel ──")
for i, s in enumerate(X_te[:5]):
    k = bot.keputusan(s)
    bot.aksi(k)
    print(f"   Benar: {iris.target_names[y_te[i]]}")
